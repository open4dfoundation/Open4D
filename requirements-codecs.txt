# The shared pip set for every Open4D codec. Installed by the root
# environment.yml; also usable on its own inside an existing Python 3.12 venv.
#
# These are direct dependencies only. The per-codec environment.yml files used to
# be `conda env export` dumps -- several hundred transitive packages pinned to
# exact versions, including things no codec imports. That is what made the
# environments impossible to reconcile: nothing recorded what was actually
# needed, only what happened to be installed.
#
# GPU-only and manually built packages are deliberately absent. See
# requirements-gpu.txt.

# --- shared by every codec ---
numpy==1.26.4
open3d==0.19.0
scipy>=1.10,<2
trimesh>=4.1
tqdm

# --- torch. On Linux with CUDA, install from the cu126 index first:
#       pip install torch==2.7.0 torchvision==0.22.0 \
#         --index-url https://download.pytorch.org/whl/cu126
#     Otherwise the CPU build below is enough for evaluation and for the codecs'
#     non-training paths.
torch==2.7.0
torchvision==0.22.0

# --- evaluation: D1/D2 PSNR, SSIM, packing (draco, klt, n4mc, tsmc) ---
